Sub punteroA(m As Integer, x As Variant, i, dc As Variant)
    Dim j As Integer, k As Integer
    Dim d1 As Variant, d2 As Variant

    k = m
10  If ((k - j) <= 1#) Then
        i = k
    Else

        End If

    End If

    d2 = Datos(i) - Datos(i - 1)
    dc = d1 / d2

End Sub
